CREATE OR REPLACE FORCE VIEW processing_lots_trans AS
   SELECT 'GREIGE-ISSUE' AS trans_type, m.cc_code, m.dated, d.party_code,
          d.contract_num, d.lot_num, d.q_code, d.g_width, c.f_width,
          c.process_type, c.design_code, c.version_code, c.colour_code,
          NULL AS machine_code, d.qnty, 0 AS ref_id
     FROM greige_issue_m m, greige_issue_d d, processing_contract_d c
    WHERE m.gi_main_id = d.gi_main_id
      AND d.pc_dtl_id = c.pc_dtl_id
      AND m.dated >= '01-SEP-26'
   UNION ALL
/* ================= PROCESSING USED ================= */
   SELECT 'PROCESS-USED' AS trans_type, l.cc_code, l.dated, l.party_code,
          l.contract_num, l.lot_num, l.q_code, l.g_width, l.f_width,
          c.process_type, c.design_code, c.version_code, c.colour_code,
          l.machine_code, -l.qnty_used AS qnty, l.processing_lot_id AS ref_id
     FROM processing_lots l, greige_issue_d g, processing_contract_d c
    WHERE l.lot_num = g.lot_num
      AND g.pc_dtl_id = c.pc_dtl_id
      AND l.dated >= '01-SEP-26'
   UNION ALL
/* ================= PROCESSING ISSUE ================= */
   SELECT 'PROCESS-PRODUCED' AS trans_type, l.to_cc_code, l.dated,
          l.party_code, l.contract_num, l.lot_num, l.q_code, l.g_width,
          NVL (l.f_width_produced, l.f_width) AS f_width, c.process_type,
          c.design_code, c.version_code, c.colour_code, l.machine_code,
          l.qnty_produced AS qnty, l.processing_lot_id AS ref_id
     FROM processing_lots l, greige_issue_d g, processing_contract_d c
    WHERE l.lot_num = g.lot_num
      AND g.pc_dtl_id = c.pc_dtl_id
      AND l.dated >= '01-SEP-26'
